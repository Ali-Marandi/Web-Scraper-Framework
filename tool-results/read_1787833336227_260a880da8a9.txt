   156→   156→    def _run_fuzzy_compare(self):
   157→   157→        s1 = self._fuzzy_s1.get().strip()
   158→   158→        s2 = self._fuzzy_s2.get().strip()
   159→   159→        if not s1 or not s2:
   160→   160→            return
   161→   161→        from core.fuzzy_engine import FuzzyMatcher
   162→   162→        m = FuzzyMatcher()
   163→   163→        fe = m.fuzzy_equal(s1, s2)
   164→   164→        jw = m.jaro_winkler(s1, s2)
   165→   165→        lev = m.levenshtein(s1, s2)
   166→   166→        fc = m.fuzzy_contains(s1, s2)
   167→   167→
   168→   168→        self._fuzzy_results.configure(state="normal")
   169→   169→        self._fuzzy_results.delete("0.0", "end")
   170→   170→        self._fuzzy_results.insert("end", f"  Fuzzy Equal Score:  {fe:.4f}
   171→   171→"
   172→   172→        self._fuzzy_results.insert("end", f"  Jaro-Winkler:       {jw:.4f}
   173→   173→"
   174→   174→        self._fuzzy_results.insert("end", f"  Levenshtein Dist:   {lev}
   175→   175→"
   176→   176→        self._fuzzy_results.insert("end", f"  Fuzzy Contains:     {fc:.4f}
   177→   177→"
   178→   178→        self._fuzzy_results.insert("end", f"  Max Length:         {max(len(s1), len(s2))}
   179→   179→"
   180→   180→        self._fuzzy_results.insert("end", f"  Normalized Lev Sim: {1.0 - (lev / max(len(s1), len(s2), 1)):.4f}
   181→   181→"
   182→   182→        self._fuzzy_results.configure(state="disabled")

   184→   184→    def _run_fuzzy_dedup(self):
   185→   185→        text = self._dedup_input.get("0.0", "end").strip()
   186→   186→        if not text:
   187→   187→            return
   188→   188→        items = [line.strip() for line in text.split("
   189→   189→") if line.strip()]
   190→   190→        try:
   191→   191→            threshold = float(self._dedup_threshold.get().strip())
   192→   192→        except ValueError:
   193→   193→            threshold = 0.85
   194→   194→        from core.fuzzy_engine import FuzzyMatcher
   195→   195→        m = FuzzyMatcher()
   196→   196→        unique = m.deduplicate(items, threshold)
   197→   197→        removed = len(items) - len(unique)
   198→   198→
   199→   199→        self._fuzzy_results.configure(state="normal")
   200→   200→        self._fuzzy_results.delete("0.0", "end")
   201→   201→        self._fuzzy_results.insert("end", f"  Input items:   {len(items)}
   202→   202→"
   203→   203→        self._fuzzy_results.insert("end", f"  Unique items:  {len(unique)}
   204→   204→"
   205→   205→        self._fuzzy_results.insert("end", f"  Duplicates:    {removed}
   206→   206→"
   207→   207→        self._fuzzy_results.insert("end", f"  Threshold:     {threshold:.2f}
   208→   208→"
   209→   209→        self._fuzzy_results.insert("end", f"
   210→   210→  --- Unique Items ---
   211→   211→")
   212→   212→        for item in unique:
   213→   213→            self._fuzzy_results.insert("end", f"  {item}

   279→   279→    def _run_content_analysis(self):
   280→   280→        url = self._content_url.get().strip()
   281→   281→        text = self._content_input.get("0.0", "end").strip()
   282→   282→
   283→   283→        if url and not text:
   284→   284→            def _fetch():
   285→   285→                try:
   286→   286→                    import requests
   287→   287→                    from bs4 import BeautifulSoup
   288→   288→                    resp = requests.get(url, timeout=15, headers={"User-Agent": "Mozilla/5.0"})
   289→   289→                    soup = BeautifulSoup(resp.text, "lxml")
   290→   290→                    for tag in soup(["script", "style", "nav", "footer", "header"]):
   291→   291→                        tag.decompose()
   292→   292→                    fetched = soup.get_text(separator=" ", strip=True)
   293→   293→                    self.after(0, lambda: self._content_input.insert("0.0", fetched))
   294→   294→                    self.after(0, lambda: self._analyze_content(fetched))
   295→   295→                except Exception as e:
   296→   296→                    self.after(0, lambda: self._show_content_error(str(e)))
   297→   297→            threading.Thread(target=_fetch, daemon=True).start()
   298→   298→        elif text:
   299→   299→            self._analyze_content(text)

   812→   812→    def _run_graph_analysis(self):
   813→   813→        text = self._graph_input.get("0.0", "end").strip()
   814→   814→        if not text:
   815→   815→            return
   816→   816→
   817→   817→        urls = [line.strip() for line in text.split("
   818→   818→") if line.strip()]
   819→   819→        if len(urls) < 2:
   820→   820→            self._graph_results.configure(state="normal")
   821→   821→            self._graph_results.delete("0.0", "end")
   822→   822→            self._graph_results.insert("end", "  Need at least 2 URLs to build a graph.
   823→   823→"
   824→   824→            self._graph_results.configure(state="disabled")
   825→   825→            return
   826→   826→
   827→   827→        from core.graph_analyzer import DiGraph, CentralityAnalyzer, CommunityDetector
   828→   828→        from urllib.parse import urlparse
   829→   829→        import numpy as np
   830→   830→
   831→   831→        self._graph_results.configure(state="normal")
   832→   832→        self._graph_results.delete("0.0", "end")

   834→   834→        try:
   835→   835→            g = DiGraph()

   846→   846→                    if other == url:
   847→   847→                        continue

   850→   850→                        g.add_edge(url, other)
