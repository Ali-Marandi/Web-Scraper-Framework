   156→    def _run_fuzzy_compare(self):
   157→        s1 = self._fuzzy_s1.get().strip()
   158→        s2 = self._fuzzy_s2.get().strip()
   159→        if not s1 or not s2:
   160→            return
   161→        from core.fuzzy_engine import FuzzyMatcher
   162→        m = FuzzyMatcher()
   163→        fe = m.fuzzy_equal(s1, s2)
   164→        jw = m.jaro_winkler(s1, s2)
   165→        lev = m.levenshtein(s1, s2)
   166→        fc = m.fuzzy_contains(s1, s2)
   167→
   168→        self._fuzzy_results.configure(state="normal")
   169→        self._fuzzy_results.delete("0.0", "end")
   170→        self._fuzzy_results.insert("end", f"  Fuzzy Equal Score:  {fe:.4f}
   171→"
   172→        self._fuzzy_results.insert("end", f"  Jaro-Winkler:       {jw:.4f}
   173→"
   174→        self._fuzzy_results.insert("end", f"  Levenshtein Dist:   {lev}
   175→"
   176→        self._fuzzy_results.insert("end", f"  Fuzzy Contains:     {fc:.4f}
   177→"
   178→        self._fuzzy_results.insert("end", f"  Max Length:         {max(len(s1), len(s2))}
   179→"
   180→        self._fuzzy_results.insert("end", f"  Normalized Lev Sim: {1.0 - (lev / max(len(s1), len(s2), 1)):.4f}
   181→"
   182→        self._fuzzy_results.configure(state="disabled")

   184→    def _run_fuzzy_dedup(self):
   185→        text = self._dedup_input.get("0.0", "end").strip()
   186→        if not text:
   187→            return
   188→        items = [line.strip() for line in text.split("
   189→") if line.strip()]
   190→        try:
   191→            threshold = float(self._dedup_threshold.get().strip())
   192→        except ValueError:
   193→            threshold = 0.85
   194→        from core.fuzzy_engine import FuzzyMatcher
   195→        m = FuzzyMatcher()
   196→        unique = m.deduplicate(items, threshold)
   197→        removed = len(items) - len(unique)
   198→
   199→        self._fuzzy_results.configure(state="normal")
   200→        self._fuzzy_results.delete("0.0", "end")
   201→        self._fuzzy_results.insert("end", f"  Input items:   {len(items)}
   202→"
   203→        self._fuzzy_results.insert("end", f"  Unique items:  {len(unique)}
   204→"
   205→        self._fuzzy_results.insert("end", f"  Duplicates:    {removed}
   206→"
   207→        self._fuzzy_results.insert("end", f"  Threshold:     {threshold:.2f}
   208→"
   209→        self._fuzzy_results.insert("end", f"
   210→  --- Unique Items ---
   211→")
   212→        for item in unique:
   213→            self._fuzzy_results.insert("end", f"  {item}
   214→"
   215→        self._fuzzy_results.configure(state="disabled")

   279→    def _run_content_analysis(self):
   280→        url = self._content_url.get().strip()
   281→        text = self._content_input.get("0.0", "end").strip()
   282→
   283→        if url and not text:
   284→            def _fetch():
   285→                try:
   286→                    import requests
   287→                    from bs4 import BeautifulSoup
   288→                    resp = requests.get(url, timeout=15, headers={"User-Agent": "Mozilla/5.0"})
   289→                    soup = BeautifulSoup(resp.text, "lxml")
   290→                    for tag in soup(["script", "style", "nav", "footer", "header"]):
   291→                        tag.decompose()
   292→                    fetched = soup.get_text(separator=" ", strip=True)
   293→                    self.after(0, lambda: self._content_input.insert("0.0", fetched))
   294→                    self.after(0, lambda: self._analyze_content(fetched))
   295→                except Exception as e:
   296→                    self.after(0, lambda: self._show_content_error(str(e)))
   297→            threading.Thread(target=_fetch, daemon=True).start()
   298→        elif text:
   299→            self._analyze_content(text)

   301→    def _analyze_content(self, text: str):
   302→        from core.content_analyzer import ContentAnalyzer, TextStatistics, SentimentAnalyzer, KeywordExtractor
   303→        self._content_results.configure(state="normal")
   304→        self._content_results.delete("0.0", "end")
   305→
   306→        try:
   307→            stats = TextStatistics()
   308→            readability = stats.compute_readability(text)
   309→            freq = stats.compute_word_frequency(text, top_n=15)
   310→            entropy = stats.compute_entropy(text)
   311→            diversity = stats.compute_lexical_diversity(text)
   312→            lang = stats.language_detect(text)
   313→
   314→            self._content_results.insert("end", "=== Text Statistics ===
   315→"
   316→            self._content_results.insert("end", f"  Words:           {readability['word_count']}
   317→"
   318→            self._content_results.insert("end", f"  Sentences:       {readability['sentence_count']}
   319→"
   320→            self._content_results.insert("end", f"  Avg Sentence:    {readability['avg_sentence_length']:.1f} words
   321→"
   322→            self._content_results.insert("end", f"  Avg Word Len:    {readability['avg_word_length']:.2f} chars
   323→"
   324→            self._content_results.insert("end", f"  Flesch-Kincaid:  {readability['flesch_kincaid']:.1f}
   325→"
   326→            self._content_results.insert("end", f"  Shannon Entropy: {entropy:.4f}
   327→"
   328→            self._content_results.insert("end", f"  Lexical Div:     {diversity:.4f}
   329→"
   330→            self._content_results.insert("end", f"  Detected Lang:   {lang}
   331→"
   332→
   333→            sa = SentimentAnalyzer()
   334→            sent = sa.analyze(text)
   335→            self._content_results.insert("end", f"
   336→=== Sentiment Analysis ===
   337→")
   338→            self._content_results.insert("end", f"  Polarity:    {sent['polarity_score']:.3f} [-1, 1]
   339→"
   340→            self._content_results.insert("end", f"  Subjectivity: {sent['subjectivity']:.3f} [0, 1]
   341→"
   342→            self._content_results.insert("end", f"  Label:       {sent['label']}
   343→"
   344→
   345→            ke = KeywordExtractor()
   346→            keywords = ke.extract_keywords_rake(text, top_n=10)
   347→            self._content_results.insert("end", f"
   348→=== Top Keywords (RAKE) ===
   349→")
   350→            for kw, score in keywords:
   351→                self._content_results.insert("end", f"  {score:.3f}  {kw}
   352→"
   353→
   354→            entities = ke.extract_named_entities_basic(text)
   355→            if entities:
   356→                self._content_results.insert("end", f"
   357→=== Named Entities ===
   358→")
   359→                for ent, etype in entities[:20]:
   360→                    self._content_results.insert("end", f"  [{etype}] {ent}
   361→"
   362→
   363→            from core.content_analyzer import ContentFingerprinter
   364→            cf = ContentFingerprinter()
   365→            summary = ContentAnalyzer().generate_summary(text, max_sentences=3)
   366→            if summary:
   367→                self._content_results.insert("end", f"
   368→=== Extractive Summary ===
   369→")
   370→                self._content_results.insert("end", f"  {summary}
   371→"
   372→
   373→        except Exception as e:
   374→            self._content_results.insert("end", f"Error: {e}
   375→"
   376→
   377→        self._content_results.configure(state="disabled")

   815→            return

   817→        urls = [line.strip() for line in text.split("

   820→            self._graph_results.configure(state="normal")
