   225→   225→    def _import_rules(self):
   226→   226→        path = filedialog.askopenfilename(
   227→   227→            filetypes=[("JSON files", "*.json"), ("All files", "*.*")],
   228→   228→        )
   229→   229→        if not path:
   230→   230→            return
   231→   231→        try:
   232→   232→            with open(path, "r", encoding="utf-8") as f:
   233→   233→                rules_data = json.load(f)
   234→   234→            count = 0
   235→   235→            for rd in rules_data if isinstance(rules_data, list) else [rules_data]:
   236→   236→                if "name" in rd and "method" in rd and "selector" in rd:
   237→   237→                    self._extraction_rules.append(ExtractionRule(
   238→   238→                        name=rd["name"], method=ExtractionMethod(rd["method"]),
   239→   239→                        selector=rd["selector"], attribute=rd.get("attribute"),
   240→   240→                        default=rd.get("default"), is_list=rd.get("is_list", False),
   241→   241→                    ))
   242→   242→                    count += 1
   243→   243→            self._refresh_rules_list()
   244→   244→            app = self.winfo_toplevel()
   245→   245→            if hasattr(app, '_log_panel'):
   246→   246→                app._log_panel.add_log(f"Imported {count} rules from {path}", "info")
   247→   247→        except Exception as e:
   248→   248→            messagebox.showerror("Import Error", str(e))
   249→   249→
   250→   250→    def _export_rules(self):
   251→   251→        if not self._extraction_rules:
   252→   252→            messagebox.showinfo("Export", "No rules to export.")
   253→   253→            return
   254→   254→        path = filedialog.asksaveasfilename(
   255→   255→            defaultextension=".json",
   256→   256→            filetypes=[("JSON files", "*.json"), ("All files", "*.*")],
   257→   257→        )
   258→   258→        if not path:
   259→   259→            return
   260→   260→        try:
   261→   261→            data = [self._rule_to_dict(r) for r in self._extraction_rules]
   262→   262→            with open(path, "w", encoding="utf-8") as f:
   263→   263→                json.dump(data, f, indent=2, ensure_ascii=False)
   264→   264→            app = self.winfo_toplevel()
   265→   265→            if hasattr(app, '_log_panel'):
   266→   266→                app._log_panel.add_log(f"Exported {len(data)} rules to {path}", "success")
   267→   267→        except Exception as e:
   268→   268→            messagebox.showerror("Export Error", str(e))

   270→   270→    def _rule_to_dict(self, rule: ExtractionRule) -> dict:

   273→   273→            "selector": rule.selector, "attribute": rule.attribute,
   274→   274→            "default": rule.default, "is_list": rule.is_list,
   275→   275→        }
