    60→    60→

    71→    71→    dimension_scores: Dict[QualityDimension, float]
    72→    72→    issues: List[QualityIssue]
    73→    73→    timestamp: float
    74→    74→    record_count: int
    75→    75→    weighted_score: float = 0.0
    76→    76→

    78→    78→    DEFAULT_WEIGHTS: Dict[QualityDimension, float] = field(
    79→    79→        default_factory=lambda: {
    80→    80→            QualityDimension.COMPLETENESS: 0.25,
    81→    81→            QualityDimension.CONSISTENCY: 0.20,
    82→    82→            QualityDimension.ACCURACY: 0.15,
    83→    83→            QualityDimension.TIMELINESS: 0.10,

    91→    91→            self.weighted_score = self._compute_weighted(self.DEFAULT_WEIGHTS)
    92→    92→
    93→    93→    def _compute_weighted(self, weights: Dict[QualityDimension, float]) -> float:
    94→    94→        total_w = sum(weights.values())
    95→    95→        if total_w == 0:
    96→    96→            return self.overall_score
    97→    97→        return sum(self.dimension_scores.get(d, 0.0) * w for d, w in weights.items()) / total_w

   105→   105→class BayesianScorer:
   106→   106→    """Bayesian quality scorer using a Beta-Bernoulli conjugate model.
   107→   107→
   108→   108→    Each observation is a Boolean (pass / fail).  The posterior is
   109→   109→    ``Beta(alpha + successes, beta + failures)`` whose expected value
   110→   110→    serves as the quality score.
   111→   111→
   112→   112→    Parameters:
   113→   113→        prior_alpha: Shape parameter for the Beta prior (default 1 → uniform).
   114→   114→        prior_beta:  Shape parameter for the Beta prior (default 1 → uniform).
   115→   115→    """
   116→   116→
   117→   117→    def __init__(self, prior_alpha: float = 1.0, prior_beta: float = 1.0) -> None:
   118→   118→        if prior_alpha <= 0 or prior_beta <= 0:
   119→   119→            raise ValueError("Prior parameters must be positive.")
   120→   120→        self._alpha: float = float(prior_alpha)
   121→   121→        self._beta: float = float(prior_beta)
   122→   122→
   123→   123→    # -- mutation --------------------------------------------------------
   124→   124→
   125→   125→    def update(self, observation: bool) -> None:
   126→   126→        """Incorporate a single Boolean observation into the posterior."""
   127→   127→        if observation:
   128→   128→            self._alpha += 1.0
   129→   129→        else:
   130→   130→            self._beta += 1.0
   131→   131→
   132→   132→    def batch_update(self, observations: List[bool]) -> None:
   133→   133→        """Incorporate a batch of Boolean observations at once."""
   134→   134→        successes = sum(1 for o in observations if o)
   135→   135→        failures = len(observations) - successes
   136→   136→        self._alpha += float(successes)
   137→   137→        self._beta += float(failures)
   138→   138→
   139→   139→    def reset(self) -> None:
   140→   140→        """Reset the posterior to the original prior."""
   141→   141→        self._alpha = 1.0
   142→   142→        self._beta = 1.0
   143→   143→
   144→   144→    # -- queries ---------------------------------------------------------
   145→   145→
   146→   146→    def get_score(self) -> float:
   147→   147→        """Return the expected value of the posterior Beta distribution.
   148→   148→
   149→   149→        Equivalent to the MAP estimate under a uniform prior and
   150→   150→        represents the current Bayesian quality score in [0, 1].
   151→   151→        """
   152→   152→        return self._alpha / (self._alpha + self._beta)
   153→   153→
   154→   154→    def get_confidence(self) -> float:
   155→   155→        """Return a confidence metric in [0, 1] based on posterior variance.
   156→   156→
   157→   157→        Confidence is defined as ``1 - sqrt(variance)`` where variance
   158→   158→        is ``alpha*beta / ((alpha+beta)^2 * (alpha+beta+1))``.
   159→   159→        """
   160→   160→        a, b = self._alpha, self._beta
   161→   161→        variance = (a * b) / ((a + b) ** 2 * (a + b + 1))
   162→   162→        return float(np.clip(1.0 - math.sqrt(variance), 0.0, 1.0))
   163→   163→
   164→   164→    def get_credible_interval(self, ci: float = 0.95) -> Tuple[float, float]:
   165→   165→        """Return an equal-tailed credible interval for the posterior.
   166→   166→
   167→   167→        Uses a numerical approximation based on the regularised
   168→   168→        incomplete Beta function (scipy-free via the CDF approximation).
   169→   169→
   170→   170→        Parameters:
   171→   171→            ci: Desired credible level (default 0.95).
   172→   172→
   173→   173→        Returns:
   174→   174→            ``(low, high)`` tuple.
   175→   175→        """
   176→   176→        from math import lgamma  # local import to stay explicit
   177→   177→
   178→   178→        a, b = self._alpha, self._beta
   179→   179→        tail = (1.0 - ci) / 2.0
   180→   180→
   181→   181→        def _beta_cdf(x: float) -> float:
   182→   182→            """Regularised incomplete Beta function I_x(a, b)."""
   183→   183→            if x <= 0.0:
   184→   184→                return 0.0
   185→   185→            if x >= 1.0:
   186→   186→                return 1.0
   187→   187→            # Lentz continued-fraction expansion
   188→   188→            lbeta = lgamma(a) + lgamma(b) - lgamma(a + b)
   189→   189→            prefix = math.exp(
   190→   190→                a * math.log(x) + b * math.log(1.0 - x) - lbeta
   191→   191→            )
   192→   192→            if x < (a + 1.0) / (a + b + 2.0):
   193→   193→                return prefix * self._betacf(a, b, x) / a
   194→   194→            return 1.0 - prefix * self._betacf(b, a, 1.0 - x) / b
   195→   195→
   196→   196→        lo = self._binsearch(_beta_cdf, tail, 0.0, 1.0)
   197→   197→        hi = self._binsearch(_beta_cdf, 1.0 - tail, 0.0, 1.0)
   198→   198→        return (lo, hi)
   199→   199→
   200→   200→    @staticmethod
   201→   201→    def _betacf(a: float, b: float, x: float, max_iter: int = 200, eps: float = 3e-12) -> float:
   202→   202→        """Continued-fraction evaluation for the incomplete Beta."""
   203→   203→        qab = a + b
   204→   204→        qap = a + 1.0
   205→   205→        qam = a - 1.0
   206→   206→        c = 1.0
   207→   207→        d = 1.0 - qab * x / qap
   208→   208→        if abs(d) < 1e-30:
   209→   209→            d = 1e-30
   210→   210→        d = 1.0 / d
   211→   211→        h = d
   212→   212→        for m in range(1, max_iter + 1):
   213→   213→            m2 = 2 * m
   214→   214→            aa = m * (b - m) * x / ((qam + m2) * (a + m2))
   215→   215→            d = 1.0 + aa * d
   216→   216→            if abs(d) < 1e-30:
   217→   217→                d = 1e-30
   218→   218→            c = 1.0 + aa / c
   219→   219→            if abs(c) < 1e-30:
   220→   220→                c = 1e-30
   221→   221→            d = 1.0 / d
   222→   222→            h *= d * c
   223→   223→            aa = -(a + m) * (qab + m) * x / ((a + m2) * (qap + m2))
   224→   224→            d = 1.0 + aa * d
   225→   225→            if abs(d) < 1e-30:
   226→   226→                d = 1e-30
   227→   227→            c = 1.0 + aa / c
   228→   228→            if abs(c) < 1e-30:
   229→   229→                c = 1e-30
   230→   230→            d = 1.0 / d
   231→   231→            delta = d * c
   232→   232→            h *= delta
   233→   233→            if abs(delta - 1.0) < eps:
   234→   234→                break
   235→   235→        return h
   236→   236→
   237→   237→    @staticmethod
   238→   238→    def _binsearch(fn: Callable[[float], float], target: float, lo: float, hi: float) -> float:
   239→   239→        """Binary search for *x* where ``fn(x) == target``."""
   240→   240→        for _ in range(60):
   241→   241→            mid = (lo + hi) / 2.0
   242→   242→            if fn(mid) < target:
   243→   243→                lo = mid
   244→   244→            else:
   245→   245→                hi = mid
   246→   246→        return (lo + hi) / 2.0
   247→   247→
   248→   248→
   249→   249→class BayesianMultivariateScorer:
   250→   250→    """Tracks multiple quality dimensions, each with its own BayesianScorer.
   251→   251→
   252→   252→    The combined score is the geometric mean of individual dimension scores,
   253→   253→    which penalises any single weak dimension.
   254→   254→    """
   255→   255→
   256→   256→    def __init__(
   257→   257→        self,
   258→   258→        dimensions: Optional[Sequence[QualityDimension]] = None,
   259→   259→        prior_alpha: float = 1.0,
   260→   260→        prior_beta: float = 1.0,
   261→   261→    ) -> None:
   262→   262→        dims = dimensions or list(QualityDimension)
   263→   263→        self._scorers: Dict[QualityDimension, BayesianScorer] = {
   264→   264→            d: BayesianScorer(prior_alpha, prior_beta) for d in dims
   265→   265→        }
   266→   266→
   267→   267→    def update(self, dimension: QualityDimension, observation: bool) -> None:
   268→   268→        """Feed a single observation for a specific dimension."""
   269→   269→        self._scorers[dimension].update(observation)
   270→   270→
   271→   271→    def batch_update(self, observations: Dict[QualityDimension, List[bool]]) -> None:
   272→   272→        """Feed batched observations per dimension."""
   273→   273→        for dim, obs_list in observations.items():
   274→   274→            self._scorers[dim].batch_update(obs_list)
   275→   275→
   276→   276→    def get_score(self, dimension: Optional[QualityDimension] = None) -> float:
   277→   277→        """Score for one dimension, or the geometric mean across all."""
   278→   278→        if dimension is not None:
   279→   279→            return self._scorers[dimension].get_score()
   280→   280→        scores = [s.get_score() for s in self._scorers.values()]
   281→   281→        # Geometric mean (add small epsilon to avoid log(0))
   282→   282→        log_sum = sum(math.log(max(s, 1e-12)) for s in scores)
   283→   283→        return math.exp(log_sum / len(scores))
   284→   284→
   285→   285→    def get_confidence(self, dimension: Optional[QualityDimension] = None) -> float:
   286→   286→        if dimension is not None:
   287→   287→            return self._scorers[dimension].get_confidence()
   288→   288→        return float(
   289→   289→            np.mean([s.get_confidence() for s in self._scorers.values()])
   290→   290→        )
   291→   291→
   292→   292→    def reset(self) -> None:
   293→   293→        for scorer in self._scorers.values():
   294→   294→            scorer.reset()
   295→   295→

   298→   298→# Information-Theory Metrics
   299→   299→# ---------------------------------------------------------------------------
   300→   300→
   301→   301→
   302→   302→class InformationTheoryMetrics:
   303→   303→    """Information-theoretic measures computed over discrete value lists.
   304→   304→
   305→   305→    All methods accept plain Python iterables and use NumPy internally
   306→   306→    for performance on larger inputs.
   307→   307→    """
   308→   308→
   309→   309→    _EPS = 1e-12
   310→   310→
   311→   311→    # -- core entropy -----------------------------------------------------
   312→   312→
   313→   313→    @staticmethod
   314→   314→    def shannon_entropy(values: Sequence[Any]) -> float:
   315→   315→        """Compute Shannon entropy H(X) in nats.
   316→   316→
   317→   317→        Parameters:
   318→   318→            values: Sequence of hashable items.
   319→   319→
   320→   320→        Returns:
   321→   321→            Entropy in nats (natural log base).  Returns 0.0 for a
   322→   322→            single-unique-value sequence.
   323→   323→        """
   324→   324→        arr = np.array(values, dtype=object)
   325→   325→        if arr.size == 0:
   326→   326→            return 0.0
   327→   327→        _, counts = np.unique(arr, return_counts=True)
   328→   328→        probs = counts / counts.sum()
   329→   329→        return float(-np.sum(probs * np.log(probs + InformationTheoryMetrics._EPS)))
   330→   330→
   331→   331→    @staticmethod
   332→   332→    def conditional_entropy(x: Sequence[Any], y: Sequence[Any]) -> float:
   333→   333→        """Compute conditional entropy H(X|Y) in nats.
   334→   334→
   335→   335→        Parameters:
   336→   336→            x: Target variable.
   337→   337→            y: Conditioning variable (same length as *x*).
   338→   338→
   339→   339→        Returns:
   340→   340→            H(X | Y) in nats.
   341→   341→        """
   342→   342→        if len(x) != len(y):
   343→   343→            raise ValueError("x and y must have the same length.")
   344→   344→        x_arr = np.array(x, dtype=object)
   345→   345→        y_arr = np.array(y, dtype=object)
   346→   346→        h = 0.0
   347→   347→        for y_val in np.unique(y_arr):
   348→   348→            mask = y_arr == y_val
   349→   349→            x_sub = x_arr[mask]
   350→   350→            h += (mask.sum() / len(x_arr)) * InformationTheoryMetrics.shannon_entropy(x_sub)
   351→   351→        return h
   352→   352→
   353→   353→    @staticmethod
   354→   354→    def mutual_information(x: Sequence[Any], y: Sequence[Any]) -> float:
   355→   355→        """Compute mutual information I(X; Y) = H(X) - H(X|Y) in nats."""
   356→   356→        return InformationTheoryMetrics.shannon_entropy(x) - InformationTheoryMetrics.conditional_entropy(x, y)
   357→   357→
   358→   358→    @staticmethod
   359→   359→    def kl_divergence(p: Sequence[float], q: Sequence[float]) -> float:
   360→   360→        """Compute Kullback-Leibler divergence D_KL(P || Q) in nats.
   361→   361→
   362→   362→        Both *p* and *q* are interpreted as probability mass functions
   363→   363→        (will be normalised if they do not sum to 1).
   364→   364→
   365→   365→        Returns:
   366→   366→            KL divergence (non-negative).  Raises if lengths differ.
   367→   367→        """
   368→   368→        if len(p) != len(q):
   369→   369→            raise ValueError("p and q must have the same length.")
   370→   370→        pa = np.asarray(p, dtype=float)
   371→   371→        qa = np.asarray(q, dtype=float)
   372→   372→        pa = pa / pa.sum()
   373→   373→        qa = qa / qa.sum()
   374→   374→        qa = np.clip(qa, InformationTheoryMetrics._EPS, None)
   375→   375→        pa = np.clip(pa, InformationTheoryMetrics._EPS, None)
   376→   376→        return float(np.sum(pa * np.log(pa / qa)))
   377→   377→
   378→   378→    # -- composite metrics ------------------------------------------------
   379→   379→
   380→   380→    @staticmethod
   381→   381→    def data_complexity_score(data: List[Dict[str, Any]]) -> float:
   382→   382→        """Composite complexity score in [0, 1].
   383→   383→
   384→   384→        Combines:
   385→   385→        * average per-field Shannon entropy (normalised by ln(n)),
   386→   386→        * cardinality ratio (unique values / total records),
   387→   387→        * sparsity (fraction of missing values).
   388→   388→
   389→   389→        Higher means more complex / information-rich data.
   390→   390→        """
   391→   391→        if not data:
   392→   392→            return 0.0
   393→   393→        n = len(data)
   394→   394→        fields = [k for k in data[0] if isinstance(data[0], dict)]
   395→   395→        if not fields:
   396→   396→            return 0.0
   397→   397→        ln_n = math.log(max(n, 2))
   398→   398→        entropy_scores: list[float] = []
   399→   399→        cardinality_scores: list[float] = []
   400→   400→        missing_total = 0
   401→   401→        total_cells = 0
   402→   402→
   403→   403→        for f in fields:
   404→   404→            col = [row.get(f) for row in data]
   405→   405→            non_empty = [v for v in col if v is not None and v != ""]
   406→   406→            missing_total += n - len(non_empty)
   407→   407→            total_cells += n
   408→   408→            if non_empty:
   409→   409→                ent = InformationTheoryMetrics.shannon_entropy(non_empty)
   410→   410→                entropy_scores.append(min(ent / ln_n, 1.0))
   411→   411→                cardinality_scores.append(
   412→   412→                    min(len(set(non_empty)) / max(n, 1), 1.0)
   413→   413→                )
   414→   414→
   415→   415→        avg_entropy = float(np.mean(entropy_scores)) if entropy_scores else 0.0
   416→   416→        avg_card = float(np.mean(cardinality_scores)) if cardinality_scores else 0.0
   417→   417→        sparsity = 1.0 - (missing_total / max(total_cells, 1))
   418→   418→
   419→   419→        return float(np.clip(0.4 * avg_entropy + 0.35 * avg_card + 0.25 * sparsity, 0.0, 1.0))
   420→   420→
   421→   421→    @staticmethod
   422→   422→    def redundancy_ratio(data: List[Dict[str, Any]]) -> float:
   423→   423→        """Fraction of records that are near-duplicates, in [0, 1].
   424→   424→
   425→   425→        Uses a SHA-256 hash of a normalised string representation of
   426→   426→        each record.  The redundancy ratio is ``1 - unique_hashes / n``.
   427→   427→        """
   428→   428→        if not data:
   429→   429→            return 0.0
   430→   430→        seen: set[str] = set()
   431→   431→        dupes = 0
   432→   432→        for row in data:
   433→   433→            norm = "|".join(f"{k}={row.get(k, '')}" for k in sorted(row))
   434→   434→            h = hashlib.sha256(norm.encode("utf-8", errors="replace")).hexdigest()
   435→   435→            if h in seen:
   436→   436→                dupes += 1
   437→   437→            else:
   438→   438→                seen.add(h)
   439→   439→        return dupes / len(data)

   459→   459→class DataQualityAssessor:
   460→   460→    """Main entry point for assessing quality of scraped records.
   461→   461→
   462→   462→    Parameters:
   463→   463→        rules:         Optional list of callables ``(record) -> str | None``
   464→   464→                       that return a description when a consistency rule is
   465→   465→                       violated (or ``None`` when the record passes).
   466→   466→        reference_data: Optional dict ``{key_field: set_of_valid_values}``
   467→   467→                       used for accuracy cross-referencing.
   468→   468→        field_weights:  Optional per-dimension weight overrides for
   469→   469→                       ``QualityReport.weighted_score``.
   470→   470→    """
   471→   471→
   472→   472→    def __init__(
   473→   473→        self,
   474→   474→        rules: Optional[List[Callable[[Dict[str, Any]], Optional[str]]]] = None,
   475→   475→        reference_data: Optional[Dict[str, set]] = None,
   476→   476→        field_weights: Optional[Dict[QualityDimension, float]] = None,
   477→   477→    ) -> None:
   478→   478→        self._rules = rules or []
   479→   479→        self._reference = reference_data or {}
   480→   480→        self._custom_weights = field_weights
   481→   481→        self._history: List[QualityReport] = []
   482→   482→        self._bayesian = BayesianMultivariateScorer()
   483→   483→
   484→   484→    # -- public API -------------------------------------------------------
   485→   485→
   486→   486→    def assess(self, data: List[Dict[str, Any]], rules: Optional[List[Callable]] = None) -> QualityReport:
   487→   487→        """Run a full quality assessment on *data* and return a report.
   488→   488→
   489→   489→        Parameters:
   490→   490→            data:  List of record dicts.
   491→   491→            rules: Override instance rules for this call only.
   492→   492→
   493→   493→        Returns:
   494→   494→            A ``QualityReport`` with per-dimension scores, issues, and
   495→   495→            aggregate metrics.
   496→   496→        """
   497→   497→        active_rules = rules if rules is not None else self._rules
   498→   498→        if not data:
   499→   499→            ts = datetime.now(timezone.utc).timestamp()
   500→   500→            empty = {d: 0.0 for d in QualityDimension}
   501→   501→            report = QualityReport(
   502→   502→                overall_score=0.0,
   503→   503→                dimension_scores=empty,
   504→   504→                issues=[],
   505→   505→                timestamp=ts,
   506→   506→                record_count=0,
   507→   507→            )
   508→   508→            self._history.append(report)
   509→   509→            return report
   510→   510→
   511→   511→        n = len(data)
   512→   512→        fields = list(data[0].keys()) if data[0] else []
   513→   513→        issues: List[QualityIssue] = []
   514→   514→
   515→   515→        # --- Completeness -------------------------------------------------
   516→   516→        missing_per_field: Dict[str, int] = {f: 0 for f in fields}
   517→   517→        for row in data:
   518→   518→            for f in fields:
   519→   519→                val = row.get(f)
   520→   520→                if val is None or (isinstance(val, str) and val.strip() == ""):
   521→   521→                    missing_per_field[f] += 1
   522→   522→
   523→   523→        completeness_scores = {
   524→   524→            f: 1.0 - (cnt / n) for f, cnt in missing_per_field.items()
   525→   525→        }
   526→   526→        completeness = float(np.mean(list(completeness_scores.values())))
   527→   527→        self._add_worst_field_issues(
   528→   528→            missing_per_field, n, QualityDimension.COMPLETENESS, issues
   529→   529→        )
   530→   530→
   531→   531→        # --- Consistency --------------------------------------------------
   532→   532→        consistency_violations = 0
   533→   533→        for row in data:
   534→   534→            for rule_fn in active_rules:
   535→   535→                msg = rule_fn(row)
   536→   536→                if msg is not None:
   537→   537→                    consistency_violations += 1
   538→   538→                    issues.append(
   539→   539→                        QualityIssue(
   540→   540→                            dimension=QualityDimension.CONSISTENCY,
   541→   541→                            field="",
   542→   542→                            description=msg,
   543→   543→                            severity="medium",
   544→   544→                            affected_count=1,
   545→   545→                        )
   546→   546→                    )
   547→   547→        consistency = 1.0 - (consistency_violations / (n * max(len(active_rules), 1)))
   548→   548→
   549→   549→        # --- Uniqueness ---------------------------------------------------
   550→   550→        redundancy = InformationTheoryMetrics.redundancy_ratio(data)
   551→   551→        uniqueness = 1.0 - redundancy
   552→   552→        dupe_count = int(redundancy * n)
   553→   553→        if dupe_count > 0:
   554→   554→            issues.append(
   555→   555→                QualityIssue(
   556→   556→                    dimension=QualityDimension.UNIQUENESS,
   557→   557→                    field="",
   558→   558→                    description=f"{dupe_count} near-duplicate record(s) detected.",
   559→   559→                    severity="high" if dupe_count > n * 0.1 else "medium",
   560→   560→                    affected_count=dupe_count,
   561→   561→                )
   562→   562→            )
   563→   563→
   564→   564→        # --- Validity -----------------------------------------------------
   565→   565→        validity_total = 0
   566→   566→        validity_pass = 0
   567→   567→        for f in fields:
   568→   568→            for row in data:
   569→   569→                val = row.get(f)
   570→   570→                if val is None or val == "":
   571→   571→                    continue
   572→   572→                validity_total += 1
   573→   573→                if self._is_valid(f, val):
   574→   574→                    validity_pass += 1
   575→   575→
   576→   576→        # also check field-name heuristics for type hints
   577→   577→        validity = validity_pass / max(validity_total, 1)
   578→   578→        invalid_count = validity_total - validity_pass
   579→   579→        if invalid_count > 0:
   580→   580→            issues.append(
   581→   581→                QualityIssue(
   582→   582→                    dimension=QualityDimension.VALIDITY,
   583→   583→                    field="(multiple)",
   584→   584→                    description=f"{invalid_count} value(s) failed format/type checks.",
   585→   585→                    severity="medium",
   586→   586→                    affected_count=invalid_count,
   587→   587→                )
   588→   588→            )
   589→   589→
   590→   590→        # --- Timeliness ---------------------------------------------------
   591→   591→        timeliness, timeliness_issues = self._assess_timeliness(data, fields)
   592→   592→        issues.extend(timeliness_issues)
   593→   593→
   594→   594→        # --- Accuracy -----------------------------------------------------
   595→   595→        accuracy, accuracy_issues = self._assess_accuracy(data)
   596→   596→        issues.extend(accuracy_issues)
   597→   597→
   598→   598→        dim_scores: Dict[QualityDimension, float] = {
   599→   599→            QualityDimension.COMPLETENESS: completeness,
   600→   600→            QualityDimension.CONSISTENCY: consistency,
   601→   601→            QualityDimension.UNIQUENESS: uniqueness,
   602→   602→            QualityDimension.VALIDITY: validity,
   603→   603→            QualityDimension.TIMELINESS: timeliness,
   604→   604→            QualityDimension.ACCURACY: accuracy,
   605→   605→        }
   606→   606→
   607→   607→        # Bayesian update for each dimension
   608→   608→        for dim, score in dim_scores.items():
   609→   609→            self._bayesian.update(dim, score >= 0.8)
   610→   610→
   611→   611→        overall = float(np.mean(list(dim_scores.values())))
   612→   612→        ts = datetime.now(timezone.utc).timestamp()
   613→   613→        report = QualityReport(
   614→   614→            overall_score=overall,
   615→   615→            dimension_scores=dim_scores,
   616→   616→            issues=issues,
   617→   617→            timestamp=ts,
   618→   618→            record_count=n,
   619→   619→        )
   620→   620→        if self._custom_weights:
   621→   621→            report.weighted_score = report._compute_weighted(self._custom_weights)
   622→   622→        self._history.append(report)
   623→   623→        return report
   624→   624→
   625→   625→    # -- time-series helpers ----------------------------------------------

   627→   627→    def track_quality_over_time(self) -> List[QualityReport]:
   628→   628→        """Return all historical reports recorded by ``assess()``."""
   629→   629→        return list(self._history)

   713→   713→    @staticmethod
   714→   714→    def _is_valid(field_name: str, value: Any) -> bool:
   715→   715→        """Heuristic format validator based on field name and value."""
   716→   716→        if not isinstance(value, str):
   717→   717→            return True  # non-string types are assumed valid
   718→   718→        fl = field_name.lower()
   719→   719→        if any(kw in fl for kw in ("url", "link", "href")):
   720→   720→            return bool(_PATTERN_URL.match(value))
   721→   721→        if "email" in fl or "mail" in fl:
   722→   722→            return bool(_PATTERN_EMAIL.match(value))
   723→   723→        if any(kw in fl for kw in ("date", "time", "created", "updated", "published")):
   724→   724→            return bool(_PATTERN_DATE.match(value))
   725→   725→        if any(kw in fl for kw in ("price", "amount", "count", "rating", "score", "age")):
   726→   726→            return bool(_PATTERN_NUM.match(value))
   727→   727→        return True

   729→   729→    @staticmethod
   730→   730→    def _assess_timeliness(
   731→   731→        data: List[Dict[str, Any]], fields: List[str]
   732→   732→    ) -> Tuple[float, List[QualityIssue]]:
   733→   733→        """Score recency when timestamp-like fields are present."""
   734→   734→        ts_fields = [
   735→   735→            f
   736→   736→            for f in fields
   737→   737→            if any(kw in f.lower() for kw in ("date", "time", "created", "updated", "published", "timestamp"))
   738→   738→        ]
   739→   739→        issues: List[QualityIssue] = []
   740→   740→        if not ts_fields:
   741→   741→            return 1.0, issues  # no temporal data → not penalised
   742→   742→
   743→   743→        now = datetime.now(timezone.utc)
   744→   744→        total_delta_days = 0.0
   745→   745→        count = 0
   746→   746→        for f in ts_fields:
   747→   747→            for row in data:
   748→   748→                val = row.get(f)
   749→   749→                if not isinstance(val, str):
   750→   750→                    continue
   751→   751→                parsed = DataQualityAssessor._parse_timestamp(val)
   752→   752→                if parsed is None:
   753→   753→                    continue
   754→   754→                delta = (now - parsed).total_seconds() / 86400.0
   755→   755→                total_delta_days += delta
   756→   756→                count += 1
   757→   757→
   758→   758→        if count == 0:
   759→   759→            return 1.0, issues
   760→   760→
   761→   761→        avg_days = total_delta_days / count
   762→   762→        # Score decays exponentially; half-life = 30 days
   763→   763→        timeliness = math.exp(-math.log(2) * avg_days / 30.0)
   764→   764→        timeliness = float(np.clip(timeliness, 0.0, 1.0))
   765→   765→
   766→   766→        if timeliness < 0.8:
   767→   767→            issues.append(
   768→   768→                QualityIssue(
   769→   769→                    dimension=QualityDimension.TIMELINESS,
   770→   770→                    field=ts_fields[0],
   771→   771→                    description=f"Average data age is {avg_days:.1f} days.",
   772→   772→                    severity="low",
   773→   773→                    affected_count=count,
   774→   774→                )
   775→   775→            )
   776→   776→        return timeliness, issues
   777→   777→
   778→   778→    def _assess_accuracy(
   779→   779→        self, data: List[Dict[str, Any]]
   780→   780→    ) -> Tuple[float, List[QualityIssue]]:
   781→   781→        """Cross-reference values against ``reference_data``."""
   782→   782→        issues: List[QualityIssue] = []
   783→   783→        if not self._reference:
   784→   784→            return 1.0, issues
   785→   785→        total = 0
   786→   786→        matched = 0
   787→   787→        for key_field, valid_values in self._reference.items():
   788→   788→            for row in data:
   789→   789→                val = row.get(key_field)
   790→   790→                if val is None:
   791→   791→                    continue
   792→   792→                total += 1
   793→   793→                if val in valid_values:
   794→   794→                    matched += 1
   795→   795→                else:
   796→   796→                    issues.append(
   797→   797→                        QualityIssue(
   798→   798→                            dimension=QualityDimension.ACCURACY,
   799→   799→                            field=key_field,
   800→   800→                            description=f"Value '{val}' not found in reference set.",
   801→   801→                            severity="high",
   802→   802→                            affected_count=1,
   803→   803→                        )
   804→   804→                    )
   805→   805→        if total == 0:
   806→   806→            return 1.0, issues
   807→   807→        return matched / total, issues

   809→   809→    @staticmethod
   810→   810→    def _add_worst_field_issues(
   811→   811→        missing: Dict[str, int],
   812→   812→        n: int,
   813→   813→        dim: QualityDimension,
   814→   814→        issues: List[QualityIssue],
   815→   815→        top_n: int = 3,
   816→   816→    ) -> None:
   817→   817→        """Append issues for the fields with the most missing values."""
   818→   818→        ranked = sorted(missing.items(), key=lambda kv: kv[1], reverse=True)
   819→   819→        for field, cnt in ranked[:top_n]:
   820→   820→            if cnt == 0:
   821→   821→                continue
   822→   822→            ratio = cnt / n
   823→   823→            sev = "critical" if ratio > 0.5 else "high" if ratio > 0.25 else "medium" if ratio > 0.1 else "low"
   824→   824→            issues.append(
   825→   825→                QualityIssue(
   826→   826→                    dimension=dim,
   827→   827→                    field=field,
   828→   828→                    description=f"{cnt}/{n} records missing value for '{field}'.",
   829→   829→                    severity=sev,
   830→   830→                    affected_count=cnt,
   831→   831→                )
   832→   832→            )

   834→   834→    @staticmethod
   835→   835→    def _parse_timestamp(s: str) -> Optional[datetime]:
   836→   836→        """Best-effort ISO-8601 timestamp parser."""
   837→   837→        for fmt in (
   838→   838→            "%Y-%m-%dT%H:%M:%S%z",
   839→   839→            "%Y-%m-%dT%H:%M:%S",
   840→   840→            "%Y-%m-%d %H:%M:%S",
   841→   841→            "%Y-%m-%d",
   842→   842→        ):
   843→   843→            try:
   844→   844→                return datetime.strptime(s, fmt)
   845→   845→            except (ValueError, TypeError):
   846→   846→                continue
   847→   847→        return None
