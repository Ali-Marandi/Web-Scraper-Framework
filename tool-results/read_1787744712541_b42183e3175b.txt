   128→    def _save_project(self):
   129→        app = self.winfo_toplevel()
   130→        if not hasattr(app, 'engine'):
   131→            return
   132→        engine = app.engine
   133→        # Gather current UI state into the project
   134→        if not engine.current_project:
   135→            engine.create_project("Untitled")
   136→        proj = engine.current_project
   137→        proj.urls = [u.strip() for u in self._url_text.get("0.0", "end").strip().splitlines() if u.strip()]
   138→        proj.mode = ScrapingMode(self._mode_seg.get().lower())
   139→        proj.extraction_rules = [self._rule_to_dict(r) for r in self._extraction_rules]
   140→        proj.page_actions = self._collect_page_actions()
   141→        proj.auto_scroll = self._var_auto_scroll.get()
   142→        proj.follow_links = self._var_follow_links.get()
   143→        try:
   144→            proj.max_pages = int(self._max_pages_entry.get().strip() or "100")
   145→        except ValueError:
   146→            proj.max_pages = 100
   147→        try:
   148→            proj.max_depth = int(self._depth_entry.get().strip() or "1")
   149→        except ValueError:
   150→            proj.max_depth = 1
   151→        engine.save_current_project()
   152→        if hasattr(app, '_log_panel'):
   153→            app._log_panel.add_log(f"Project saved: {proj.name}", "success")

   202→    def _apply_project_to_ui(self, proj):
   203→        self._url_text.configure(state="normal")
   204→        self._url_text.delete("0.0", "end")
   205→        for url in proj.urls:
   206→            self._url_text.insert("end", url + "\n")
   207→
   208→        self._mode_seg.set(proj.mode.value.capitalize())
   209→        self._var_auto_scroll.set(proj.auto_scroll)
   210→        self._var_follow_links.set(proj.follow_links)
   211→        self._max_pages_entry.delete("0", "end")
   212→        self._max_pages_entry.insert("0", str(proj.max_pages))
   213→        self._depth_entry.delete("0", "end")
   214→        self._depth_entry.insert("0", str(proj.max_depth))
   215→
   216→        self._extraction_rules.clear()
   217→        for rd in proj.extraction_rules:
   218→            self._extraction_rules.append(ExtractionRule(
   219→                name=rd["name"], method=ExtractionMethod(rd["method"]),
   220→                selector=rd["selector"], attribute=rd.get("attribute"),
   221→                default=rd.get("default"), is_list=rd.get("is_list", False),
   222→            ))
   223→        self._refresh_rules_list()

   225→    def _import_rules(self):
   226→        path = filedialog.askopenfilename(
   227→            filetypes=[("JSON files", "*.json"), ("All files", "*.*")],
   228→        )
   229→        if not path:
   230→            return
   231→        try:
   232→            with open(path, "r", encoding="utf-8") as f:
   233→                rules_data = json.load(f)
   234→            count = 0
   235→            for rd in rules_data if isinstance(rules_data, list) else [rules_data]:
   236→                if "name" in rd and "method" in rd and "selector" in rd:
   237→                    self._extraction_rules.append(ExtractionRule(
   238→                        name=rd["name"], method=ExtractionMethod(rd["method"]),
   239→                        selector=rd["selector"], attribute=rd.get("attribute"),
   240→                        default=rd.get("default"), is_list=rd.get("is_list", False),
   241→                    ))
   242→                    count += 1
   243→            self._refresh_rules_list()
   244→            app = self.winfo_toplevel()
   245→            if hasattr(app, '_log_panel'):
   246→                app._log_panel.add_log(f"Imported {count} rules from {path}", "info")
   247→        except Exception as e:
   248→            messagebox.showerror("Import Error", str(e))
   249→
   250→    def _export_rules(self):
   251→        if not self._extraction_rules:
   252→            messagebox.showinfo("Export", "No rules to export.")
   253→            return
   254→        path = filedialog.asksaveasfilename(
   255→            defaultextension=".json",
   256→            filetypes=[("JSON files", "*.json"), ("All files", "*.*")],
   257→        )
   258→        if not path:
   259→            return
   260→        try:
   261→            data = [self._rule_to_dict(r) for r in self._extraction_rules]
   262→            with open(path, "w", encoding="utf-8") as f:
   263→                json.dump(data, f, indent=2, ensure_ascii=False)
   264→            app = self.winfo_toplevel()
   265→            if hasattr(app, '_log_panel'):
   266→                app._log_panel.add_log(f"Exported {len(data)} rules to {path}", "success")
   267→        except Exception as e:
   268→            messagebox.showerror("Export Error", str(e))

   270→    def _rule_to_dict(self, rule: ExtractionRule) -> dict:
   271→        return {
   272→            "name": rule.name, "method": rule.method.value,
   273→            "selector": rule.selector, "attribute": rule.attribute,
   274→            "default": rule.default, "is_list": rule.is_list,
   275→        }

   387→    def _refresh_rules_list(self):
   388→        self._rules_listbox.configure(state="normal")
   389→        self._rules_listbox.delete("0.0", "end")
   390→        for i, rule in enumerate(self._extraction_rules):
   391→            list_flag = " [LIST]" if rule.is_list else ""
   392→            attr = f" attr={rule.attribute}" if rule.attribute else ""
   393→            default = f" default=\"{rule.default}\"" if rule.default else ""
   394→            line = f"{i + 1}. {rule.name}  |  {rule.method.value}  |  {rule.selector}{attr}{default}{list_flag}\n"
   395→            self._rules_listbox.insert("end", line)
   396→        self._rules_listbox.configure(state="disabled")

   401→    def _edit_rule(self):
   402→        sel = self._rules_listbox.get("1.0", "end").strip().split("\n")
   403→        if not sel:
   404→            return
   405→        idx = int(self._rules_listbox.index("insert").split(".")[0]) - 1
   406→        if 0 <= idx < len(self._extraction_rules):
   407→            self._show_rule_dialog(edit_index=idx)

   409→    def _remove_rule(self):

   411→            self._extraction_rules.pop()

   414→    def _clear_rules(self):
   415→        self._extraction_rules.clear()

   724→    def _on_start(self):
   725→        app = self.winfo_toplevel()
   726→        if not hasattr(app, 'engine'):
   727→            return
   728→        engine = app.engine
   729→        urls_text = self._url_text.get("0.0", "end").strip()
   730→        if not urls_text:
   731→            messagebox.showwarning("Warning", "Please enter at least one URL.")
   732→            return
   733→        urls = [u.strip() for u in urls_text.splitlines() if u.strip()]
   734→        valid_urls = []
   735→        for u in urls:
   736→            if u.startswith(("http://", "https://")):
   737→                valid_urls.append(u)
   738→            else:
   739→                messagebox.showwarning("Invalid URL", f"URL must start with http:// or https://:\n{u}")
   740→                return
   741→        if not valid_urls:
   742→            return
   743→        mode_map = {"Static": ScrapingMode.STATIC, "Dynamic": ScrapingMode.DYNAMIC, "Auto": ScrapingMode.AUTO}
   744→        mode = mode_map.get(self._mode_seg.get(), ScrapingMode.STATIC)
   745→        engine.data_parser.clear_rules()
   746→        for rule in self._extraction_rules:
   747→            engine.data_parser.add_rule(rule)
   748→        engine.clear_results()
   749→        self._btn_start.configure(state="disabled")
   750→        self._btn_stop.configure(state="normal")
   751→        if hasattr(app, '_log_panel'):
   752→            app._log_panel.add_log(f"Starting {mode.value} scrape of {len(valid_urls)} URL(s)", "info")
   753→        engine.scrape_urls(valid_urls, mode, callback=self._on_progress)
   754→
   755→    def _on_stop(self):
   756→        app = self.winfo_toplevel()
   757→        if hasattr(app, 'engine') and app.engine:
   758→            app.engine.stop()
   759→            if hasattr(app, '_log_panel'):
   760→                app._log_panel.add_log("Scraping stopped by user", "warning")
   761→        self._btn_start.configure(state="normal")
   762→        self._btn_stop.configure(state="disabled")
   763→
   764→    def _on_progress(self, current, total, records):
