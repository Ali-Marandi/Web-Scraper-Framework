   189→   189→    def _refresh_rules_list(self):
   190→   190→        self._rules_listbox.configure(state="normal")
   191→   191→        self._rules_listbox.delete("0.0", "end")
   192→   192→        for i, rule in enumerate(self._extraction_rules):
   193→   193→            list_flag = " [LIST]" if rule.is_list else ""
   194→   194→            attr = f" attr={rule.attribute}" if rule.attribute else ""
   195→   195→            default = f" default=\"{rule.default}\"" if rule.default else ""
   196→   196→            line = f"{i + 1}. {rule.name}  |  {rule.method.value}  |  {rule.selector}{attr}{default}{list_flag}\n"
   197→   197→            self._rules_listbox.insert("end", line)
   198→   198→        self._rules_listbox.configure(state="disabled")

   203→   203→    def _edit_rule(self):
   204→   204→        sel = self._rules_listbox.get("1.0", "end").strip().split("\n")
   205→   205→        if not sel:
   206→   206→            return
   207→   207→        idx = int(self._rules_listbox.index("insert").split(".")[0]) - 1
   208→   208→        if 0 <= idx < len(self._extraction_rules):
   209→   209→            self._show_rule_dialog(edit_index=idx)

   211→   211→    def _remove_rule(self):
   212→   212→        if self._extraction_rules:
   213→   213→            self._extraction_rules.pop()
   214→   214→            self._refresh_rules_list()

   430→   430→    def _display_results(self, results: list[dict]):
   431→   431→        self._results_text.configure(state="normal")
   432→   432→        self._results_text.delete("0.0", "end")
   433→   433→        if not results:
   434→   434→            self._results_text.insert("end", "No results yet. Add extraction rules and start scraping.")
   435→   435→        else:
   436→   436→            # Filter out internal fields
   437→   437→            display = []
   438→   438→            for r in results:
   439→   439→                display.append({k: v for k, v in r.items() if not k.startswith("_")})
   440→   440→            headers = list(display[0].keys()) if display else []
   441→   441→            col_widths = {}
   442→   442→            for h in headers:
   443→   443→                w = len(str(h))
   444→   444→                for r in display:
   445→   445→                    val = str(r.get(h, ""))
   446→   446→                    if isinstance(r.get(h), list):
   447→   447→                        val = "; ".join(str(v) for v in r.get(h, []))
   448→   448→                    w = max(w, len(val))
   449→   449→                col_widths[h] = min(w, 40)
   450→   450→
   451→   451→            header_line = "  |  ".join(str(h).ljust(col_widths.get(h, 12)) for h in headers)
   452→   452→            sep_line = "-+-".join("-" * col_widths.get(h, 12) for h in headers)
   453→   453→            self._results_text.insert("end", header_line + "\n")
   454→   454→            self._results_text.insert("end", sep_line + "\n")
   455→   455→            for row in display:
   456→   456→                cells = []
   457→   457→                for h in headers:
   458→   458→                    val = row.get(h, "")
   459→   459→                    if isinstance(val, list):
   460→   460→                        val = "; ".join(str(v) for v in val)
   461→   461→                    cells.append(str(val)[:col_widths.get(h, 12)].ljust(col_widths.get(h, 12)))
   462→   462→                self._results_text.insert("end", "  |  ".join(cells) + "\n")
   463→   463→        self._results_text.configure(state="disabled")

   530→   530→    def _on_start(self):
   531→   531→        app = self.winfo_toplevel()
   532→   532→        if not hasattr(app, 'engine'):
   533→   533→            return
   534→   534→        engine = app.engine
   535→   535→        urls_text = self._url_text.get("0.0", "end").strip()
   536→   536→        if not urls_text:
   537→   537→            messagebox.showwarning("Warning", "Please enter at least one URL.")
   538→   538→            return
   539→   539→        urls = [u.strip() for u in urls_text.splitlines() if u.strip()]
   540→   540→        # Basic URL validation
   541→   541→        valid_urls = []
   542→   542→        for u in urls:
   543→   543→            if u.startswith(("http://", "https://")):
   544→   544→                valid_urls.append(u)
   545→   545→            else:
   546→   546→                messagebox.showwarning("Invalid URL", f"URL must start with http:// or https://:\n{u}")
   547→   547→                return
   548→   548→        if not valid_urls:
   549→   549→            return
   550→   550→        mode_map = {"Static": ScrapingMode.STATIC, "Dynamic": ScrapingMode.DYNAMIC, "Auto": ScrapingMode.AUTO}
   551→   551→        mode = mode_map.get(self._mode_seg.get(), ScrapingMode.STATIC)
   552→   552→
   553→   553→        # Clear old rules and add current
   554→   554→        engine.data_parser.clear_rules()
   555→   555→        for rule in self._extraction_rules:
   556→   556→            engine.data_parser.add_rule(rule)
   557→   557→
   558→   558→        engine.clear_results()
   559→   559→        self._btn_start.configure(state="disabled")
   560→   560→        self._btn_stop.configure(state="normal")
   561→   561→        if hasattr(app, '_log_panel'):
   562→   562→            app._log_panel.add_log(f"Starting {mode.value} scrape of {len(valid_urls)} URL(s)", "info")
   563→   563→        engine.scrape_urls(valid_urls, mode, callback=self._on_progress)
   564→   564→
