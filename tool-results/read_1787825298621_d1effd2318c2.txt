   148→   148→    def _run_regex_test(self):
   149→   149→        pattern = self._regex_pattern.get().strip()
   150→   150→        text = self._regex_input.get("0.0", "end").strip()
   151→   151→        if not pattern or not text:
   152→   152→            return
   153→   153→
   154→   154→        flags = 0
   155→   155→        if self._regex_flags.get("i") and self._regex_flags["i"].get(): flags |= re.IGNORECASE
   156→   156→        if self._regex_flags.get("m") and self._regex_flags["m"].get(): flags |= re.MULTILINE
   157→   157→        if self._regex_flags.get("s") and self._regex_flags["s"].get(): flags |= re.DOTALL
   158→   158→        if self._regex_flags.get("x") and self._regex_flags["x"].get(): flags |= re.VERBOSE
   159→   159→
   160→   160→        try:
   161→   161→            matches = re.findall(pattern, text, flags)
   162→   162→            self._regex_results.configure(state="normal")
   163→   163→            self._regex_results.delete("0.0", "end")
   164→   164→            if not matches:
   165→   165→                self._regex_results.insert("end", "No matches found.")
   166→   166→                self._match_count_label.configure(text="0 matches")
   167→   167→            else:
   168→   168→                for i, m in enumerate(matches[:500]):
   169→   169→                    if isinstance(m, tuple):
   170→   170→                        self._regex_results.insert("end", f"{i+1}. {m}\n")
   171→   171→                    else:
   172→   172→                        self._regex_results.insert("end", f"{i+1}. {m}\n")
   173→   173→                count = len(matches)
   174→   174→                self._match_count_label.configure(text=f"{count} match{'es' if count != 1 else ''}")
   175→   175→            self._regex_results.configure(state="disabled")
   176→   176→        except re.error as e:
   177→   177→            self._regex_results.configure(state="normal")
   178→   178→            self._regex_results.delete("0.0", "end")
   179→   179→            self._regex_results.insert("end", f"Regex error: {e}")
   180→   180→            self._regex_results.configure(state="disabled")
   181→   181→            self._match_count_label.configure(text="Error")

   268→   268→    def _load_html_for_css(self):
   269→   269→        url = self._css_html_input.get("0.0", "end").strip()[:200]
   270→   270→        if url.startswith(("http://", "https://")):
   271→   271→            self._fetch_url_html(url, "css")
   272→   272→        else:
   273→   273→            from tkinter import filedialog
   274→   274→            path = filedialog.askopenfilename(filetypes=[("HTML", "*.html"), ("All", "*.*")])
   275→   275→            if path:
   276→   276→                try:
   277→   277→                    with open(path, "r", encoding="utf-8") as f:
   278→   278→                        content = f.read()
   279→   279→                    self._css_html_input.delete("0.0", "end")
   280→   280→                    self._css_html_input.insert("0.0", content[:100000])
   281→   281→                    self._cached_html = content
   282→   282→                except Exception as e:
   283→   283→                    messagebox = __import__("tkinter").messagebox
   284→   284→                    messagebox.showerror("Error", str(e))

   286→   286→    def _run_css_test(self):
   287→   287→        selector = self._css_selector.get().strip()
   288→   288→        html_text = self._css_html_input.get("0.0", "end").strip()
   289→   289→        attr = self._css_attr.get().strip() or None
   290→   290→
   291→   291→        if not selector or not html_text:
   292→   292→            return
   293→   293→
   294→   294→        try:
   295→   295→            soup = BeautifulSoup(html_text, "lxml")
   296→   296→            elements = soup.select(selector)
   297→   297→
   298→   298→            self._css_results.configure(state="normal")
   299→   299→            self._css_results.delete("0.0", "end")
   300→   300→
   301→   301→            if not elements:
   302→   302→                self._css_results.insert("end", "No elements matched.")
   303→   303→            else:
   304→   304→                for i, el in enumerate(elements[:200]):
   305→   305→                    if attr:
   306→   306→                        val = el.get(attr, "")
   307→   307→                        self._css_results.insert("end", f"{i+1}. [{attr}] {str(val)[:200]}\n")
   308→   308→                    else:
   309→   309→                        text = el.get_text(strip=True)
   310→   310→                        tag = el.name
   311→   311→                        classes = el.get("class", [])
   312→   312→                        cls_str = "." + ".".join(classes) if classes else ""
   313→   313→                        self._css_results.insert("end", f"{i+1}. <{tag}{cls_str}> {text[:200]}\n")
   314→   314→
   315→   315→            self._css_count_label.configure(text=f"{len(elements)} elements")
   316→   316→            self._css_results.configure(state="disabled")
   317→   317→        except Exception as e:
   318→   318→            self._css_results.configure(state="normal")
   319→   319→            self._css_results.delete("0.0", "end")
   320→   320→            self._css_results.insert("end", f"Error: {e}")
   321→   321→            self._css_results.configure(state="disabled")
   322→   322→            self._css_count_label.configure(text="Error")

   503→   503→    def _run_jsonpath_test(self):
   504→   504→        path_expr = self._jsonpath_expr.get().strip()
   505→   505→        json_text = self._json_input.get("0.0", "end").strip()
   506→   506→        if not path_expr or not json_text:
   507→   507→            return
   508→   508→
   509→   509→        try:
   510→   510→            data = json.loads(json_text)
   511→   511→        except json.JSONDecodeError as e:
   512→   512→            self._jsonpath_results.configure(state="normal")
   513→   513→            self._jsonpath_results.delete("0.0", "end")
   514→   514→            self._jsonpath_results.insert("end", f"JSON parse error: {e}")
   515→   515→            self._jsonpath_results.configure(state="disabled")
   516→   516→            return
   517→   517→
   518→   518→        try:
   519→   519→            # Simple JSON path resolver (dotted + bracket notation)
   520→   520→            result = self._resolve_json_path(data, path_expr)
   521→   521→
   522→   522→            self._jsonpath_results.configure(state="normal")
   523→   523→            self._jsonpath_results.delete("0.0", "end")
   524→   524→
   525→   525→            if result is None:
   526→   526→                self._jsonpath_results.insert("end", "Path not found.")
   527→   527→            else:
   528→   528→                output = json.dumps(result, indent=2, ensure_ascii=False, default=str)
   529→   529→                if len(output) > 50000:
   530→   530→                    output = output[:50000] + "\n\n... (truncated)"
   531→   531→                self._jsonpath_results.insert("end", output)
   532→   532→
   533→   533→            self._jsonpath_results.configure(state="disabled")
   534→   534→        except Exception as e:
   535→   535→            self._jsonpath_results.configure(state="normal")
   536→   536→            self._jsonpath_results.delete("0.0", "end")
   537→   537→            self._jsonpath_results.insert("end", f"Path error: {e}")
   538→   538→            self._jsonpath_results.configure(state="disabled")

   540→   540→    def _resolve_json_path(self, data, path: str):
   541→   541→        parts = re.split(r'\.|\[|\]', path)
   542→   542→        parts = [p for p in parts if p]
   543→   543→        current = data
   544→   544→        for part in parts:
   545→   545→            if not part:
   546→   546→                continue
   547→   547→            # Handle slice notation like [0:3]
   548→   548→            slice_match = re.match(r'^(\d+):(\d*)$', part)
   549→   549→            if slice_match and isinstance(current, list):
   550→   550→                start = int(slice_match.group(1))
   551→   551→                end = int(slice_match.group(2)) if slice_match.group(2) else len(current)
   552→   552→                current = current[start:end]
   553→   553→                continue
   554→   554→            if part.isdigit():
   555→   555→                idx = int(part)
   556→   556→                if isinstance(current, list) and idx < len(current):
   557→   557→                    current = current[idx]
   558→   558→                else:
   559→   559→                    return None
   560→   560→            elif isinstance(current, dict):
   561→   561→                current = current.get(part)
   562→   562→            else:
   563→   563→                return None
   564→   564→            if current is None:
   565→   565→                return None
   566→   566→        return current

   656→   656→    def _fetch_response(self):
   657→   657→        url = self._resp_url.get().strip()
   658→   658→        if not url:
   659→   659→            return
   660→   660→        if not url.startswith(("http://", "https://")):
   661→   661→            self._resp_status.configure(text="Invalid URL", text_color=theme.colors.TEXT_ERROR)
   662→   662→            return
   663→   663→
   664→   664→        self._resp_status.configure(text="Fetching...", text_color=theme.colors.TEXT_WARNING)
   665→   665→
   666→   666→        def _do():
   667→   667→            import requests
   668→   668→            try:
   669→   669→                start = time.time()
   670→   670→                resp = requests.get(url, timeout=15, headers={
   671→   671→                    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
   672→   672→                })
   673→   673→                elapsed = time.time() - start
   674→   674→
   675→   675→                def _update():
   676→   676→                    # Status line
   677→   677→                    size = len(resp.content)
   678→   678→                    size_str = f"{size:,} B" if size < 1024 else f"{size/1024:.1f} KB"
   679→   679→                    self._resp_status.configure(
   680→   680→                        text=f"{resp.status_code} | {elapsed:.2f}s | {size_str} | {resp.headers.get('Content-Type', '')[:40]}",
   681→   681→                        text_color=theme.colors.TEXT_SUCCESS if resp.status_code < 400 else theme.colors.TEXT_ERROR,
   682→   682→                    )
   683→   683→
   684→   684→                    # Formatted headers
   685→   685→                    self._resp_headers_text.configure(state="normal")
   686→   686→                    self._resp_headers_text.delete("0.0", "end")
   687→   687→                    self._resp_headers_text.insert("end", f"  Status:     {resp.status_code} {resp.reason}\n")
   688→   688→                    self._resp_headers_text.insert("end", f"  URL:        {resp.url}\n")
   689→   689→                    self._resp_headers_text.insert("end", f"  Time:       {elapsed:.3f}s\n")
   690→   690→                    self._resp_headers_text.insert("end", f"  Size:       {size_str}\n")

   692→   692→                    self._resp_headers_text.insert("end", f"\n  Response Headers:\n")

   696→   696→                    self._resp_headers_text.configure(state="disabled")
