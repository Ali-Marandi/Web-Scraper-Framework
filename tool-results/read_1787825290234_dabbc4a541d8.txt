   148→    def _run_regex_test(self):
   149→        pattern = self._regex_pattern.get().strip()
   150→        text = self._regex_input.get("0.0", "end").strip()
   151→        if not pattern or not text:
   152→            return
   153→
   154→        flags = 0
   155→        if self._regex_flags.get("i") and self._regex_flags["i"].get(): flags |= re.IGNORECASE
   156→        if self._regex_flags.get("m") and self._regex_flags["m"].get(): flags |= re.MULTILINE
   157→        if self._regex_flags.get("s") and self._regex_flags["s"].get(): flags |= re.DOTALL
   158→        if self._regex_flags.get("x") and self._regex_flags["x"].get(): flags |= re.VERBOSE
   159→
   160→        try:
   161→            matches = re.findall(pattern, text, flags)
   162→            self._regex_results.configure(state="normal")
   163→            self._regex_results.delete("0.0", "end")
   164→            if not matches:
   165→                self._regex_results.insert("end", "No matches found.")
   166→                self._match_count_label.configure(text="0 matches")
   167→            else:
   168→                for i, m in enumerate(matches[:500]):
   169→                    if isinstance(m, tuple):
   170→                        self._regex_results.insert("end", f"{i+1}. {m}\n")
   171→                    else:
   172→                        self._regex_results.insert("end", f"{i+1}. {m}\n")
   173→                count = len(matches)
   174→                self._match_count_label.configure(text=f"{count} match{'es' if count != 1 else ''}")
   175→            self._regex_results.configure(state="disabled")
   176→        except re.error as e:
   177→            self._regex_results.configure(state="normal")
   178→            self._regex_results.delete("0.0", "end")
   179→            self._regex_results.insert("end", f"Regex error: {e}")
   180→            self._regex_results.configure(state="disabled")
   181→            self._match_count_label.configure(text="Error")

   268→    def _load_html_for_css(self):
   269→        url = self._css_html_input.get("0.0", "end").strip()[:200]
   270→        if url.startswith(("http://", "https://")):
   271→            self._fetch_url_html(url, "css")
   272→        else:
   273→            from tkinter import filedialog
   274→            path = filedialog.askopenfilename(filetypes=[("HTML", "*.html"), ("All", "*.*")])
   275→            if path:
   276→                try:
   277→                    with open(path, "r", encoding="utf-8") as f:
   278→                        content = f.read()
   279→                    self._css_html_input.delete("0.0", "end")
   280→                    self._css_html_input.insert("0.0", content[:100000])
   281→                    self._cached_html = content
   282→                except Exception as e:
   283→                    messagebox = __import__("tkinter").messagebox
   284→                    messagebox.showerror("Error", str(e))

   286→    def _run_css_test(self):
   287→        selector = self._css_selector.get().strip()
   288→        html_text = self._css_html_input.get("0.0", "end").strip()
   289→        attr = self._css_attr.get().strip() or None
   290→
   291→        if not selector or not html_text:
   292→            return
   293→
   294→        try:
   295→            soup = BeautifulSoup(html_text, "lxml")
   296→            elements = soup.select(selector)
   297→
   298→            self._css_results.configure(state="normal")
   299→            self._css_results.delete("0.0", "end")
   300→
   301→            if not elements:
   302→                self._css_results.insert("end", "No elements matched.")
   303→            else:
   304→                for i, el in enumerate(elements[:200]):
   305→                    if attr:
   306→                        val = el.get(attr, "")
   307→                        self._css_results.insert("end", f"{i+1}. [{attr}] {str(val)[:200]}\n")
   308→                    else:
   309→                        text = el.get_text(strip=True)
   310→                        tag = el.name
   311→                        classes = el.get("class", [])
   312→                        cls_str = "." + ".".join(classes) if classes else ""
   313→                        self._css_results.insert("end", f"{i+1}. <{tag}{cls_str}> {text[:200]}\n")
   314→
   315→            self._css_count_label.configure(text=f"{len(elements)} elements")
   316→            self._css_results.configure(state="disabled")
   317→        except Exception as e:
   318→            self._css_results.configure(state="normal")
   319→            self._css_results.delete("0.0", "end")
   320→            self._css_results.insert("end", f"Error: {e}")
   321→            self._css_results.configure(state="disabled")
   322→            self._css_count_label.configure(text="Error")

   503→    def _run_jsonpath_test(self):
   504→        path_expr = self._jsonpath_expr.get().strip()
   505→        json_text = self._json_input.get("0.0", "end").strip()
   506→        if not path_expr or not json_text:
   507→            return
   508→
   509→        try:
   510→            data = json.loads(json_text)
   511→        except json.JSONDecodeError as e:
   512→            self._jsonpath_results.configure(state="normal")
   513→            self._jsonpath_results.delete("0.0", "end")
   514→            self._jsonpath_results.insert("end", f"JSON parse error: {e}")
   515→            self._jsonpath_results.configure(state="disabled")
   516→            return
   517→
   518→        try:
   519→            # Simple JSON path resolver (dotted + bracket notation)
   520→            result = self._resolve_json_path(data, path_expr)
   521→
   522→            self._jsonpath_results.configure(state="normal")
   523→            self._jsonpath_results.delete("0.0", "end")
   524→
   525→            if result is None:
   526→                self._jsonpath_results.insert("end", "Path not found.")
   527→            else:
   528→                output = json.dumps(result, indent=2, ensure_ascii=False, default=str)
   529→                if len(output) > 50000:
   530→                    output = output[:50000] + "\n\n... (truncated)"
   531→                self._jsonpath_results.insert("end", output)
   532→
   533→            self._jsonpath_results.configure(state="disabled")
   534→        except Exception as e:
   535→            self._jsonpath_results.configure(state="normal")
   536→            self._jsonpath_results.delete("0.0", "end")
   537→            self._jsonpath_results.insert("end", f"Path error: {e}")
   538→            self._jsonpath_results.configure(state="disabled")
   539→
   540→    def _resolve_json_path(self, data, path: str):
   541→        parts = re.split(r'\.|\[|\]', path)
   542→        parts = [p for p in parts if p]
   543→        current = data
   544→        for part in parts:
   545→            if not part:
   546→                continue
   547→            # Handle slice notation like [0:3]
   548→            slice_match = re.match(r'^(\d+):(\d*)$', part)
   549→            if slice_match and isinstance(current, list):
   550→                start = int(slice_match.group(1))
   551→                end = int(slice_match.group(2)) if slice_match.group(2) else len(current)
   552→                current = current[start:end]
   553→                continue
   554→            if part.isdigit():
   555→                idx = int(part)
   556→                if isinstance(current, list) and idx < len(current):
   557→                    current = current[idx]
   558→                else:
   559→                    return None
   560→            elif isinstance(current, dict):
   561→                current = current.get(part)
   562→            else:
   563→                return None
   564→            if current is None:
   565→                return None
   566→        return current

   656→    def _fetch_response(self):
   657→        url = self._resp_url.get().strip()
   658→        if not url:
   659→            return
   660→        if not url.startswith(("http://", "https://")):
   661→            self._resp_status.configure(text="Invalid URL", text_color=theme.colors.TEXT_ERROR)
   662→            return
   663→
   664→        self._resp_status.configure(text="Fetching...", text_color=theme.colors.TEXT_WARNING)
   665→
   666→        def _do():
   667→            import requests
   668→            try:
   669→                start = time.time()
   670→                resp = requests.get(url, timeout=15, headers={
   671→                    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
   672→                })
   673→                elapsed = time.time() - start
   674→
   675→                def _update():
   676→                    # Status line
   677→                    size = len(resp.content)
   678→                    size_str = f"{size:,} B" if size < 1024 else f"{size/1024:.1f} KB"
   679→                    self._resp_status.configure(
   680→                        text=f"{resp.status_code} | {elapsed:.2f}s | {size_str} | {resp.headers.get('Content-Type', '')[:40]}",
   681→                        text_color=theme.colors.TEXT_SUCCESS if resp.status_code < 400 else theme.colors.TEXT_ERROR,
   682→                    )
   683→
   684→                    # Formatted headers
   685→                    self._resp_headers_text.configure(state="normal")
   686→                    self._resp_headers_text.delete("0.0", "end")
   687→                    self._resp_headers_text.insert("end", f"  Status:     {resp.status_code} {resp.reason}\n")
   688→                    self._resp_headers_text.insert("end", f"  URL:        {resp.url}\n")
   689→                    self._resp_headers_text.insert("end", f"  Time:       {elapsed:.3f}s\n")
   690→                    self._resp_headers_text.insert("end", f"  Size:       {size_str}\n")
   691→                    self._resp_headers_text.insert("end", f"  Encoding:   {resp.encoding or 'unknown'}\n")
   692→                    self._resp_headers_text.insert("end", f"\n  Response Headers:\n")
   693→                    self._resp_headers_text.insert("end", f"  {'':-<50}\n")
   694→                    for key, val in resp.headers.items():
   695→                        self._resp_headers_text.insert("end", f"  {key}: {val}\n")
   696→                    self._resp_headers_text.configure(state="disabled")
   697→
   698→                    # Raw headers
   699→                    self._resp_raw_text.configure(state="normal")
   700→                    self._resp_raw_text.delete("0.0", "end")
   701→                    self._resp_raw_text.insert("end", f"  Request Headers:\n")
   702→                    self._resp_raw_text.insert("end", f"  {'':-<50}\n")
   703→                    for key, val in resp.request.headers.items():
   704→                        self._resp_raw_text.insert("end", f"  {key}: {val}\n")
   705→                    self._resp_raw_text.insert("end", f"\n  Response Headers:\n")
   706→                    self._resp_raw_text.insert("end", f"  {'':-<50}\n")
   707→                    for key, val in resp.headers.items():
   708→                        self._resp_raw_text.insert("end", f"  {key}: {val}\n")
   709→                    self._resp_raw_text.configure(state="disabled")
   710→
   711→                    # Body
   712→                    self._resp_body_text.configure(state="normal")
   713→                    self._resp_body_text.delete("0.0", "end")
   714→                    body = resp.text[:50000] if resp.text else "(empty body)"
   715→                    self._resp_body_text.insert("0.0", body)
   716→                    if len(resp.text or "") > 50000:
   717→                        self._resp_body_text.insert("end", f"\n\n... truncated ({len(resp.text or '')} chars total)")
   718→                    self._resp_body_text.configure(state="disabled")
   719→
   720→                    # Cache for other tools
   721→                    self._cached_html = resp.text or ""
   722→                    self._cached_url = url
   723→
   724→                self.after(0, _update)
   725→
   726→            except Exception as e:
   727→                self.after(0, lambda: self._resp_status.configure(
   728→                    text=f"Error: {str(e)[:60]}", text_color=theme.colors.TEXT_ERROR))
   729→
   730→        threading.Thread(target=_do, daemon=True).start()
