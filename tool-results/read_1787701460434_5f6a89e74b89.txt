   174→    def _refresh_rules_list(self):
   175→        self._rules_listbox.configure(state="normal")
   176→        self._rules_listbox.delete("0.0", "end")
   177→        for i, rule in enumerate(self._extraction_rules):
   178→            list_flag = " [LIST]" if rule.is_list else ""
   179→            attr = f" attr={rule.attribute}" if rule.attribute else ""
   180→            default = f" default=\"{rule.default}\"" if rule.default else ""
   181→            line = f"{i + 1}. {rule.name}  |  {rule.method.value}  |  {rule.selector}{attr}{default}{list_flag}\n"
   182→            self._rules_listbox.insert("end", line)
   183→        self._rules_listbox.configure(state="disabled")
   184→
   185→    def _add_rule(self):
   186→        self._show_rule_dialog()
   187→
   188→    def _edit_rule(self):
   189→        sel = self._rules_listbox.get("1.0", "end").strip().split("\n")
   190→        if not sel:
   191→            return
   192→        idx = int(self._rules_listbox.index("insert").split(".")[0]) - 1
   193→        if 0 <= idx < len(self._extraction_rules):
   194→            self._show_rule_dialog(edit_index=idx)
   195→
   196→    def _remove_rule(self):
   197→        if self._extraction_rules:
   198→            self._extraction_rules.pop()
   199→            self._refresh_rules_list()

   201→    def _show_rule_dialog(self, edit_index: int = -1):

   203→        dialog.title("Edit Extraction Rule" if edit_index >= 0 else "Add Extraction Rule")

   207→        dialog.grab_set()

   209→        dialog.grid_columnconfigure(1, weight=1)

   211→        existing = self._extraction_rules[edit_index] if edit_index >= 0 else None

   235→        selector_entry = ctk.CTkEntry(dialog, font=(Typography.MONO_FONT, Typography.SMALL_SIZE),

   240→        attr_entry = ctk.CTkEntry(dialog, font=(Typography.MONO_FONT, Typography.SMALL_SIZE),
   241→                                   fg_color=theme.colors.BG_INPUT, border_color=theme.colors.BORDER,
   242→                                   border_width=1, corner_radius=Radius.MD, text_color=theme.colors.TEXT_PRIMARY)
   243→        attr_entry.grid(row=3, column=1, sticky="ew", padx=(0, Spacing.MD), pady=Spacing.SM)

   247→                                      border_width=1, corner_radius=Radius.MD, text_color=theme.colors.TEXT_PRIMARY)

   357→    def _collect_page_actions(self) -> list[dict]:
   358→        actions = []
   359→        for row in self._action_rows_frame.winfo_children():
   360→            if isinstance(row, ctk.CTkFrame):
   361→                widgets = row.winfo_children()
   362→                if len(widgets) >= 3:
   363→                    action_type = widgets[0].get()
   364→                    selector = widgets[1].get().strip()
   365→                    value = widgets[2].get().strip()
   366→                    actions.append({
   367→                        "action_type": action_type,
   368→                        "selector": selector,
   369→                        "value": value,
   370→                        "delay": 0.5,
   371→                    })
   372→        return actions

   402→            headers = list(results[0].keys())
   403→            col_widths = {h: max(len(str(h)), max((len(str(r.get(h, ""))) for r in results), default=0)) for h in headers}
   404→            col_widths = {h: min(w, 40) for h, w in col_widths.items()}
   405→
   406→            header_line = "  |  ".join(str(h).ljust(col_widths.get(h, 12)) for h in headers)
   407→            sep_line = "-+-".join("-" * col_widths.get(h, 12) for h in headers)
   408→            self._results_text.insert("end", header_line + "\n")
   409→            self._results_text.insert("end", sep_line + "\n")
   410→            for row in results:
   411→                line = "  |  ".join(str(row.get(h, ""))[:col_widths.get(h, 12)].ljust(col_widths.get(h, 12)) for h in headers)
   412→                self._results_text.insert("end", line + "\n")
   413→        self._results_text.configure(state="disabled")

   460→    def _on_start(self):
   461→        engine = self.winfo_toplevel().engine
   462→        if not engine:
   463→            return
   464→        urls_text = self._url_text.get("0.0", "end").strip()
   465→        if not urls_text:
   466→            return
   467→        urls = [u.strip() for u in urls_text.splitlines() if u.strip()]
   468→        mode_map = {"Static": ScrapingMode.STATIC, "Dynamic": ScrapingMode.DYNAMIC, "Auto": ScrapingMode.AUTO}
   469→        mode = mode_map.get(self._mode_seg.get(), ScrapingMode.STATIC)
   470→
   471→        for rule in self._extraction_rules:
   472→            engine.data_parser.add_rule(rule)
   473→
   474→        self._btn_start.configure(state="disabled")
   475→        self._btn_stop.configure(state="normal")
   476→        engine.scrape_urls(urls, mode, callback=self._on_progress)
   477→
   478→    def _on_stop(self):
   479→        engine = self.winfo_toplevel().engine
   480→        if engine:
   481→            engine.stop()
   482→        self._btn_start.configure(state="normal")
   483→        self._btn_stop.configure(state="disabled")

   485→    def _on_progress(self, current, total, records):
   486→        engine = self.winfo_toplevel().engine
   487→        if engine:
   488→            self._display_results(engine.results)
   489→        if engine and engine.state.value == "idle":
   490→            self._btn_start.configure(state="normal")
   491→            self._btn_stop.configure(state="disabled")
   492→            self._display_results(engine.results)
