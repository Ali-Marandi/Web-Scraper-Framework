   174→   174→    def _refresh_rules_list(self):
   175→   175→        self._rules_listbox.configure(state="normal")
   176→   176→        self._rules_listbox.delete("0.0", "end")
   177→   177→        for i, rule in enumerate(self._extraction_rules):
   178→   178→            list_flag = " [LIST]" if rule.is_list else ""
   179→   179→            attr = f" attr={rule.attribute}" if rule.attribute else ""
   180→   180→            default = f" default=\"{rule.default}\"" if rule.default else ""
   181→   181→            line = f"{i + 1}. {rule.name}  |  {rule.method.value}  |  {rule.selector}{attr}{default}{list_flag}\n"
   182→   182→            self._rules_listbox.insert("end", line)
   183→   183→        self._rules_listbox.configure(state="disabled")

   189→   189→        sel = self._rules_listbox.get("1.0", "end").strip().split("\n")

   191→   191→            return
   192→   192→        idx = int(self._rules_listbox.index("insert").split(".")[0]) - 1

   194→   194→            self._show_rule_dialog(edit_index=idx)

   396→   396→    def _display_results(self, results: list[dict]):
   397→   397→        self._results_text.configure(state="normal")
   398→   398→        self._results_text.delete("0.0", "end")
   399→   399→        if not results:
   400→   400→            self._results_text.insert("end", "No results yet. Configure rules and start scraping.")
   401→   401→        else:
   402→   402→            headers = list(results[0].keys())
   403→   403→            col_widths = {h: max(len(str(h)), max((len(str(r.get(h, ""))) for r in results), default=0)) for h in headers}
   404→   404→            col_widths = {h: min(w, 40) for h, w in col_widths.items()}
   405→   405→
   406→   406→            header_line = "  |  ".join(str(h).ljust(col_widths.get(h, 12)) for h in headers)
   407→   407→            sep_line = "-+-".join("-" * col_widths.get(h, 12) for h in headers)
   408→   408→            self._results_text.insert("end", header_line + "\n")
   409→   409→            self._results_text.insert("end", sep_line + "\n")
   410→   410→            for row in results:
   411→   411→                line = "  |  ".join(str(row.get(h, ""))[:col_widths.get(h, 12)].ljust(col_widths.get(h, 12)) for h in headers)
   412→   412→                self._results_text.insert("end", line + "\n")
   413→   413→        self._results_text.configure(state="disabled")

   460→   460→    def _on_start(self):
   461→   461→        engine = self.winfo_toplevel().engine
   462→   462→        if not engine:
   463→   463→            return
   464→   464→        urls_text = self._url_text.get("0.0", "end").strip()

   468→   468→        mode_map = {"Static": ScrapingMode.STATIC, "Dynamic": ScrapingMode.DYNAMIC, "Auto": ScrapingMode.AUTO}

   474→   474→        self._btn_start.configure(state="disabled")
